}
bool equalsIgnoreCase(const char* a, const char* b) {
  if (!a || !b) return false;
  return strcasecmp(a, b) == 0;
}
JsonVariantConst getMember(JsonObjectConst obj, const char* target) {
  for (JsonPairConst pair : obj) {
    if (equalsIgnoreCase(pair.key().c_str(), target)) {
      return pair.value();
    }
  }
  return JsonVariantConst();
}
float jsonToFloat(JsonVariantConst value) {
  if (value.isNull()) return NAN;
  if (value.is<float>() || value.is<int>() || value.is<double>()) {
    float result = value.as<float>();
    if (isfinite(result)) return result;
  }
  if (value.is<const char*>()) {
    const char* str = value.as<const char*>();
    if (str && str[0] != '\0') {
      char* endPtr;
      float result = strtof(str, &endPtr);
      if (endPtr != str && isfinite(result)) return result;
    }
  }
  return NAN;
}
float findSensorRecursive(JsonVariantConst node, const char* targetName, int depth = 0) {
  if (depth > 15) return NAN;
  if (node.is<JsonObjectConst>()) {
    JsonObjectConst obj = node.as<JsonObjectConst>();
    JsonVariantConst nameValue = getMember(obj, "name");
    if (nameValue.isNull()) nameValue = getMember(obj, "sensorName");
    if (nameValue.is<const char*>()) {
      if (equalsIgnoreCase(nameValue.as<const char*>(), targetName)) {
        JsonVariantConst value = getMember(obj, "value");
        if (value.isNull()) value = getMember(obj, "ValueFormatted");
        float result = jsonToFloat(value);
        if (isfinite(result)) return result;
      }
    }
    for (JsonPairConst pair : obj) {
      float result = findSensorRecursive(pair.value(), targetName, depth + 1);
      if (isfinite(result)) return result;
    }
  } else if (node.is<JsonArrayConst>()) {
    for (JsonVariantConst item : node.as<JsonArrayConst>()) {
      float result = findSensorRecursive(item, targetName, depth + 1);
      if (isfinite(result)) return result;
    }
  }
  return NAN;
}
float findRootValue(JsonDocument& doc, const char* name) {
  return jsonToFloat(getMember(doc.as<JsonObjectConst>(), name));
}
bool fetchData() {
  if (WiFi.status() != WL_CONNECTED) {
    if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
      sharedData.wifiConnected = false;
      sharedData.mcuOnline     = false;
      sharedData.pcOnline      = false;
      if (sharedData.lcdKnown) {
        sharedData.lcdKnown = false;
        sharedData.lcdSeq++;
      }
      sharedData.sampleSeq++;
      xSemaphoreGive(dataMutex);
    }
    return false;
  }
  HTTPClient http;
  http.setTimeout(2000);
  http.begin(API_URL);
  int httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK) {
    Serial.printf("HTTP Error: %d\n", httpCode);
    http.end();
    if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
      sharedData.wifiConnected = true;
      sharedData.mcuOnline     = false;
      sharedData.pcOnline      = false;
      if (sharedData.lcdKnown) {
        sharedData.lcdKnown = false;
        sharedData.lcdSeq++;
      }
      sharedData.sampleSeq++;
      xSemaphoreGive(dataMutex);
    }
    return false;
  }
  JsonDocument doc;
  DeserializationError error = deserializeJson(doc, http.getStream());
  http.end();
  if (error) {
    Serial.printf("JSON Error: %s\n", error.c_str());
    if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
      sharedData.wifiConnected = true;
      sharedData.mcuOnline     = false;
      sharedData.pcOnline      = false;
      sharedData.sampleSeq++;
      xSemaphoreGive(dataMutex);
    }
    return false;
  }
  float newCase  = findRootValue(doc, "ambientTempC");
  float newPower = findRootValue(doc, "realPowerWatts");
  float ageMs    = findRootValue(doc, "lastPcDataAgeMs");
  float newCPU   = findSensorRecursive(doc.as<JsonVariantConst>(), "Core Max");
  float newGPU   = findSensorRecursive(doc.as<JsonVariantConst>(), "GPU Core");
  bool pcIsLive = isfinite(ageMs) && (ageMs <= 5000.0f);
  if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
    sharedData.wifiConnected = true;
    sharedData.caseTemp      = newCase;
    sharedData.cpuTemp       = pcIsLive ? newCPU   : NAN;
    sharedData.gpuTemp       = pcIsLive ? newGPU   : NAN;
    sharedData.pcPower       = pcIsLive ? newPower : NAN;
    sharedData.mcuOnline     = true;
    sharedData.pcOnline      = pcIsLive;
    sharedData.sampleSeq++;
    xSemaphoreGive(dataMutex);
  }
  return true;
}
bool sendPcPowerCommand() {
  if (WiFi.status() != WL_CONNECTED) return false;
  HTTPClient http;
  http.setTimeout(2500);
  http.begin(POWER_URL);
  http.addHeader("Content-Type", "application/json");
  int code = http.POST("{\"press\":true}");
  http.end();

  Serial.printf("POST /pc/power -> HTTP %d\n", code);
  return (code == 202 || code == 200);
}
bool fetchLcdStatus() {
  if (WiFi.status() != WL_CONNECTED) return false;
  HTTPClient http;
  http.setTimeout(2000);
  http.begin(LCD_STATUS_URL);
  int httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK) {
    http.end();
    return false;
  }
  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, http.getStream());
  http.end();
  if (err || !doc["lcdEnabled"].is<bool>()) {
    return false;
  }
  bool enabled = doc["lcdEnabled"].as<bool>();
  if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
    if (!sharedData.lcdKnown || sharedData.lcdEnabled != enabled) {
      sharedData.lcdKnown   = true;
      sharedData.lcdEnabled = enabled;
      sharedData.lcdSeq++;
    }
    xSemaphoreGive(dataMutex);
  }
  return true;
}
bool sendLcdControlCommand(bool targetState, bool* outEnabled) {
  if (WiFi.status() != WL_CONNECTED) return false;
  HTTPClient http;
  http.setTimeout(2500);
  http.begin(LCD_CONTROL_URL);
  http.addHeader("Content-Type", "application/json");
  const char* body = targetState ? "{\"enabled\":true}" : "{\"enabled\":false}";
  int code = http.POST(body);
  if (code != HTTP_CODE_OK) {
    http.end();
    Serial.printf("POST /lcd/control -> HTTP %d\n", code);
    return false;
  }
  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, http.getStream());
  http.end();
  if (err || !(doc["ok"] | false) || !doc["lcdEnabled"].is<bool>()) {
    return false;
  }
  *outEnabled = doc["lcdEnabled"].as<bool>();
  Serial.printf("POST /lcd/control -> HTTP %d (lcdEnabled=%s)\n",
                code, *outEnabled ? "true" : "false");
  return true;
}
// 6. BACKGROUND NETWORK TASK (CORE 0)
void networkTask(void* param) {
  uint32_t lastWiFiCheck = 0;
  uint32_t lastFetch     = 0;
  uint32_t lastLcdFetch  = 0;
  bool ntpStarted        = false;
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (true) {
    uint32_t now = millis();
    bool wifiOk = (WiFi.status() == WL_CONNECTED);
    if (wifiOk && !ntpStarted) {
      ntpStarted = true;
      configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC,
                 "pool.ntp.org", "time.google.com", "time.nist.gov");
      Serial.println("NTP UTC+7 sync started");
    }
    if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
      sharedData.wifiConnected = wifiOk;
      xSemaphoreGive(dataMutex);
    }
    bool doPower   = false;
    bool doLcd     = false;
    bool lcdTarget = false;
    if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
      if (sharedData.powerCmdRequested) {
        sharedData.powerCmdRequested = false;
        doPower = true;
      }
      if (sharedData.lcdCmdRequested) {
        sharedData.lcdCmdRequested = false;
        lcdTarget = sharedData.lcdCmdTarget;
        doLcd = true;
      }
      xSemaphoreGive(dataMutex);
    }
    if (doPower) {
      bool ok = sendPcPowerCommand();
      if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
        sharedData.powerCmdResult = ok ? CMD_OK : CMD_ERR;
        sharedData.powerCmdSeq++;
        xSemaphoreGive(dataMutex);
      }
    }
    if (doLcd) {
      bool actualEnabled = lcdTarget;
      bool ok = sendLcdControlCommand(lcdTarget, &actualEnabled);
      if (xSemaphoreTake(dataMutex, portMAX_DELAY)) {
        sharedData.lcdCmdResult = ok ? CMD_OK : CMD_ERR;
        if (ok) {
          sharedData.lcdKnown   = true;
          sharedData.lcdEnabled = actualEnabled;
          sharedData.lcdSeq++;
        }
        sharedData.lcdCmdSeq++;
        xSemaphoreGive(dataMutex);
      }
      lastLcdFetch = millis();
    }
    if (!wifiOk && (now - lastWiFiCheck >= WIFI_RETRY_INTERVAL)) {
      lastWiFiCheck = now;
      Serial.println("WiFi reconnecting...");
      WiFi.disconnect();
      WiFi.begin(WIFI_SSID, WIFI_PASS);
    }
    if (wifiOk && (lastFetch == 0 || (now - lastFetch >= REFRESH_INTERVAL))) {
      lastFetch = now;
      bool ok = fetchData();
      if (!ok) {
        lastLcdFetch = 0;
      }
    }
    if (wifiOk && (lastLcdFetch == 0 || (now - lastLcdFetch >= LCD_REFRESH_INTERVAL))) {
      lastLcdFetch = now;
      fetchLcdStatus();
    }
    vTaskDelay(pdMS_TO_TICKS(40));
  }
}

void my_disp_flush(lv_disp_drv_t* disp, const lv_area_t* area, lv_color_t* color_p) {
  uint32_t w = (area->x2 - area->x1 + 1);
  uint32_t h = (area->y2 - area->y1 + 1);
  if (lcd.getStartCount() == 0) {
    lcd.startWrite();
  }
#if LV_COLOR_16_SWAP
  lcd.pushImageDMA(area->x1, area->y1, w, h, (lgfx::swap565_t*)&color_p->full);
#else
  lcd.pushImageDMA(area->x1, area->y1, w, h, (lgfx::rgb565_t*)&color_p->full);
#endif
  lv_disp_flush_ready(disp);
}
void my_touchpad_read(lv_indev_drv_t* indev_driver, lv_indev_data_t* data) {
  static bool     lastPressed = false;
  static uint16_t lastX       = 0;
  static uint16_t lastY       = 0;
  static uint32_t lastTouchMs = 0;
  uint16_t tx = 0, ty = 0;
  bool touched = lcd.getTouch(&tx, &ty);
  if (!touched && lastPressed) {
    touched = lcd.getTouch(&tx, &ty);
  }
  uint32_t now = millis();
  if (touched) {
    tx = (uint16_t)constrain((int)tx, 0, 479);
    ty = (uint16_t)constrain((int)ty, 0, 319);
    if (lastPressed) {
      if (abs((int)tx - (int)lastX) <= 2 && abs((int)ty - (int)lastY) <= 2) {
        tx = lastX;
        ty = lastY;
      }
    }
    lastPressed = true;
    lastX       = tx;
    lastY       = ty;
    lastTouchMs = now;
    data->state   = LV_INDEV_STATE_PR;
    data->point.x = tx;
    data->point.y = ty;
#if DEBUG_TOUCH
    static uint32_t lastTouchLog = 0;
    if (now - lastTouchLog > 200) {
      lastTouchLog = now;
      Serial.printf("Touch: X=%u Y=%u\n", tx, ty);
    }
#endif
  } else if (lastPressed && (now - lastTouchMs < 18)) {
    data->state   = LV_INDEV_STATE_PR;
    data->point.x = lastX;
    data->point.y = lastY;
  } else {
    lastPressed   = false;
    data->state   = LV_INDEV_STATE_REL;
    data->point.x = lastX;
    data->point.y = lastY;
  }
}

void setArcAnimated(lv_obj_t* arc, int32_t targetVal) {
  int32_t cur = lv_arc_get_value(arc);
  if (cur == targetVal) return;
  lv_anim_del(arc, arcAnimExecCb);
  lv_anim_t a;
  lv_anim_init(&a);
  lv_anim_set_var(&a, arc);
  lv_anim_set_exec_cb(&a, arcAnimExecCb);
  lv_anim_set_time(&a, 320);
  lv_anim_set_values(&a, cur, targetVal);
  lv_anim_set_path_cb(&a, lv_anim_path_ease_out);
  lv_anim_start(&a);
}

void applyCardTheme(int idx, int lvl) {
  GaugeCard& gc = ui.cards[idx];
  if (gc.currentStatus == lvl) return;
  gc.currentStatus = lvl;
  lv_color_t bg, border, track, valCol;
  const char* statusText;
  switch (lvl) {
    case STATUS_NORMAL:
      bg = themeBg(); border = themeBorder(); track = themeTrack(); valCol = themeVal();
      statusText = "NORMAL";
      break;
    case STATUS_WARNING:
      bg = COL_WARN_BG; border = COL_WARN_BORDER; track = COL_WARN_TRACK; valCol = COL_WARN_VAL;
      statusText = "WARNING";
      break;
    case STATUS_CRITICAL:
      bg = COL_CRIT_BG; border = COL_CRIT_BORDER; track = COL_CRIT_TRACK; valCol = COL_CRIT_VAL;
      statusText = "CRITICAL";
      break;
    default:
      bg = COL_NONE_BG; border = COL_NONE_BORDER; track = COL_NONE_TRACK; valCol = COL_NONE_VAL;
      statusText = "NONE";
      break;
  }
  lv_obj_set_style_bg_color(gc.card, bg, 0);
  lv_obj_set_style_border_color(gc.card, border, 0);
  lv_obj_set_style_bg_color(gc.dot, valCol, 0);
  lv_obj_set_style_arc_color(gc.arc, track, LV_PART_MAIN);
  lv_obj_set_style_arc_color(gc.arc, valCol, LV_PART_INDICATOR);
  lv_obj_set_style_text_color(gc.valLbl, valCol, 0);
  lv_obj_set_style_bg_color(gc.alertPill, track, 0);
  lv_obj_set_style_border_color(gc.alertPill, border, 0);
  lv_obj_set_style_text_color(gc.alertLbl, valCol, 0);
  lv_label_set_text(gc.alertLbl, statusText);
}

void updateGaugeCard(int idx, float value, float warnThresh, float critThresh) {
  GaugeCard& gc = ui.cards[idx];
  if (!isfinite(value)) {
    applyCardTheme(idx, STATUS_NONE);
    if (strcmp(gc.lastText, "--.-") != 0) {
      strcpy(gc.lastText, "--.-");
      lv_label_set_text(gc.valLbl, "--.-");
      lv_obj_align_to(gc.valLbl, gc.arc, LV_ALIGN_CENTER, 0, -4);
    }
    setArcAnimated(gc.arc, gc.minRange);
    return;
  }
  int lvl = STATUS_NORMAL;
  if (value >= critThresh) lvl = STATUS_CRITICAL;
  else if (value >= warnThresh) lvl = STATUS_WARNING;
  applyCardTheme(idx, lvl);
  char buf[16];
  snprintf(buf, sizeof(buf), "%.1f", value);
  if (strcmp(gc.lastText, buf) != 0) {
    strcpy(gc.lastText, buf);
    lv_label_set_text(gc.valLbl, buf);
    lv_obj_align_to(gc.valLbl, gc.arc, LV_ALIGN_CENTER, 0, -4);
  }
  int32_t clamped = constrain((int32_t)roundf(value), gc.minRange, gc.maxRange);
  setArcAnimated(gc.arc, clamped);
}

void updateColorPresetsUI() {
  for (uint8_t i = 0; i < PRESET_COUNT; i++) {
    bool match = false;
    if (COLOR_PRESETS[i].sat == 0 && currentSat == 0) {
      match = true;
    } else if (COLOR_PRESETS[i].sat > 0 && currentSat > 0) {
      int diff = abs((int)currentHue - (int)COLOR_PRESETS[i].hue);
      if (diff > 180) diff = 360 - diff;
      if (diff <= 5) match = true;
    }
    lv_obj_set_style_border_color(ui.swatchBtns[i], match ? COL_TEXT : COL_BORDER_BASE, 0);
    lv_obj_set_style_border_width(ui.swatchBtns[i], match ? 2 : 1, 0);
    if (match) {
      lv_obj_clear_flag(ui.swatchChecks[i], LV_OBJ_FLAG_HIDDEN);
    } else {
      lv_obj_add_flag(ui.swatchChecks[i], LV_OBJ_FLAG_HIDDEN);
    }
  }
}

void applyThemeToUI() {
  lv_color_t bg  = themeBg();
  lv_color_t brd = themeBorder();
  lv_color_t val = themeVal();
  // 1. ⌘ Button
  lv_obj_set_style_bg_color(ui.iconBox, bg, 0);
  lv_obj_set_style_bg_color(ui.iconBox, brd, LV_STATE_PRESSED);
  lv_obj_set_style_border_color(ui.iconBox, brd, 0);
  for (int i = 0; i < 5; i++) {
    if (ui.iconParts[i]) {
      lv_obj_set_style_border_color(ui.iconParts[i], val, 0);
    }
  }
  // 2. Tab LIVE / CHART
  switchTab(isChartTab);
  // 3. POWER & LCD Button
  if (powerFeedbackUntil == 0) {
    setPowerBtnVisual(CMD_IDLE);
  }
  if (!lcdSending && lcdFeedbackUntil == 0) {
    setLcdBtnVisual(CMD_IDLE, uiLcdKnown, uiLcdEnabled);
  }
  // 4. PC ONLINE State
  if (lastConnState == 2) {
    lv_obj_set_style_bg_color(ui.statusDot, val, 0);
    lv_obj_set_style_text_color(ui.statusLbl, val, 0);
  }
  // 5. Normal
  for (int i = 0; i < 4; i++) {
    if (ui.cards[i].currentStatus == STATUS_NORMAL) {
      ui.cards[i].currentStatus = STATUS_INIT;
      applyCardTheme(i, STATUS_NORMAL);
    }
  }
  // 6. Power box
  lv_obj_set_style_border_color(ui.powerDlg, brd, 0);
  lv_obj_set_style_bg_color(ui.btnPowerConfirm, bg, 0);
  lv_obj_set_style_border_color(ui.btnPowerConfirm, brd, 0);
  lv_obj_set_style_text_color(ui.lblPowerConfirm, val, 0);
  // 7. Setting
  applyThemePreviewToModal();
}

// 10. CẬP NHẬT SPLASH, ĐỒNG HỒ & UI CHÍNH
void updateSplash() {
  if (bootDone) return;
  bool wifiOk = false;
  bool mcuOk  = false;
  if (xSemaphoreTake(dataMutex, 0) == pdTRUE) {
    wifiOk = sharedData.wifiConnected;
    mcuOk  = sharedData.mcuOnline;
    xSemaphoreGive(dataMutex);
  }
  uint32_t now     = millis();
  uint32_t elapsed = now - bootStartMs;
  int stage = 0;
  int32_t pct = 10;
  if (mcuOk) {
    stage = 3;
    pct = 100;
    if (bootCompleteMs == 0) bootCompleteMs = now;
  } else if (elapsed >= BOOT_TIMEOUT_MS) {
    stage = 2;
    pct = 100;
    if (bootCompleteMs == 0) bootCompleteMs = now;
  } else if (wifiOk) {
    stage = 1;
    pct = min(88, 45 + (int)(elapsed / 120));
  } else {
    stage = 0;
    pct = min(40, 10 + (int)(elapsed / 140));
  }
  if (pct != lastBootPct) {
    lastBootPct = pct;
    lv_bar_set_value(ui.bootBar, pct, LV_ANIM_ON);
  }
  if (stage != lastBootStage) {
    lastBootStage = stage;
    if (stage == 3) {
      lv_label_set_text(ui.bootStatusLbl, "Connected! Launching dashboard...");
      lv_obj_set_style_text_color(ui.bootStatusLbl, themeVal(), 0);
    } else if (stage == 2) {
      lv_label_set_text(ui.bootStatusLbl, "Offline mode - Entering dashboard...");
      lv_obj_set_style_text_color(ui.bootStatusLbl, COL_WARN_VAL, 0);
    } else if (stage == 1) {
      lv_label_set_text(ui.bootStatusLbl, "Connecting to 192.168.1.115...");
    } else {
      lv_label_set_text(ui.bootStatusLbl, "Connecting to Wi-Fi...");
    }
  }
  if (bootCompleteMs != 0 && (now - bootCompleteMs >= 350)) {
    lv_obj_add_flag(ui.pageSplash, LV_OBJ_FLAG_HIDDEN);
    bootDone = true;
    gaugeSweepAtMs = now + 60;
  }
}
void updateClock() {
  if (!bootDone) return;
  if (!lv_obj_has_flag(ui.powerModal, LV_OBJ_FLAG_HIDDEN) ||
      !lv_obj_has_flag(ui.brightModal, LV_OBJ_FLAG_HIDDEN)) {
    return;
  }
  time_t nowSec = time(nullptr);
  if (nowSec <= 1700000000) return;
  if (nowSec == lastClockSec) return;
  lastClockSec = nowSec;
  struct tm tmInfo;
  localtime_r(&nowSec, &tmInfo);
  char buf[16];
  snprintf(buf, sizeof(buf), "%02d:%02d:%02d",
           tmInfo.tm_hour, tmInfo.tm_min, tmInfo.tm_sec);
  lv_label_set_text(ui.clockLbl, buf);
}
void updateUI() {
  uint32_t now = millis();
  if (gaugeSweepAtMs != 0 && now >= gaugeSweepAtMs) {
    gaugeSweepAtMs = 0;
    lastRenderedSeq = 0;
  }
  if (powerFeedbackUntil != 0 && now >= powerFeedbackUntil) {
    powerFeedbackUntil = 0;
    setPowerBtnVisual(CMD_IDLE);
  }
  if (lcdFeedbackUntil != 0 && now >= lcdFeedbackUntil) {
    lcdFeedbackUntil = 0;
    if (!lcdSending) {
      setLcdBtnVisual(CMD_IDLE, uiLcdKnown, uiLcdEnabled);
    }
  }
  if (!lv_obj_has_flag(ui.powerModal, LV_OBJ_FLAG_HIDDEN) ||
      !lv_obj_has_flag(ui.brightModal, LV_OBJ_FLAG_HIDDEN)) {
    return;
  }
  TelemetryData snap;
  bool sensorChanged = false;
  bool powerChanged  = false;
  bool lcdChanged    = false;
  bool lcdCmdChanged = false;
  if (xSemaphoreTake(dataMutex, 0) != pdTRUE) return;
  if (sharedData.sampleSeq != lastRenderedSeq) {
    sensorChanged = true;
    lastRenderedSeq = sharedData.sampleSeq;
  }
  if (sharedData.powerCmdSeq != lastPowerSeq) {
    powerChanged = true;
    lastPowerSeq = sharedData.powerCmdSeq;
  }
  if (sharedData.lcdSeq != lastLcdSeq) {
    lcdChanged = true;
    lastLcdSeq = sharedData.lcdSeq;
  }
  if (sharedData.lcdCmdSeq != lastLcdCmdSeq) {
    lcdCmdChanged = true;
    lastLcdCmdSeq = sharedData.lcdCmdSeq;
  }
  if (sensorChanged || powerChanged || lcdChanged || lcdCmdChanged) {
    snap = sharedData;
  }
  xSemaphoreGive(dataMutex);
  if (powerChanged) {
    setPowerBtnVisual(snap.powerCmdResult);
    powerFeedbackUntil = now + 1800;
  }
  if (lcdChanged) {
    uiLcdKnown   = snap.lcdKnown;
    uiLcdEnabled = snap.lcdEnabled;
    if (!lcdSending && lcdFeedbackUntil == 0) {
      setLcdBtnVisual(CMD_IDLE, uiLcdKnown, uiLcdEnabled);
    }
  }
  if (lcdCmdChanged) {
    lcdSending   = false;
    uiLcdKnown   = snap.lcdKnown;
    uiLcdEnabled = snap.lcdEnabled;
    if (snap.lcdCmdResult == CMD_ERR) {
      setLcdBtnVisual(CMD_ERR, uiLcdKnown, uiLcdEnabled);
      lcdFeedbackUntil = now + 2500;
    } else {
      lcdFeedbackUntil = 0;
      setLcdBtnVisual(CMD_IDLE, uiLcdKnown, uiLcdEnabled);
    }
  }
  if (!sensorChanged) return;
  int connState = (snap.mcuOnline && snap.pcOnline) ? 2 : (snap.mcuOnline ? 1 : 0);
  if (connState != lastConnState) {
    lastConnState = connState;
    if (connState == 2) {
      lv_obj_set_style_bg_color(ui.statusDot, themeVal(), 0);
      lv_obj_set_style_text_color(ui.statusLbl, themeVal(), 0);
      lv_label_set_text(ui.statusLbl, "PC ONLINE");
    } else if (connState == 1) {
      lv_obj_set_style_bg_color(ui.statusDot, COL_WARN_VAL, 0);
      lv_obj_set_style_text_color(ui.statusLbl, COL_WARN_VAL, 0);
      lv_label_set_text(ui.statusLbl, "PC OFFLINE");
    } else {
      lv_obj_set_style_bg_color(ui.statusDot, COL_CRIT_VAL, 0);
      lv_obj_set_style_text_color(ui.statusLbl, COL_CRIT_VAL, 0);
      lv_label_set_text(ui.statusLbl, "MCU OFFLINE");
    }
  }
  if (bootDone) {
    updateGaugeCard(CARD_CPU,  snap.cpuTemp,  TEMP_YELLOW,    TEMP_RED);
    updateGaugeCard(CARD_GPU,  snap.gpuTemp,  TEMP_YELLOW,    TEMP_RED);
    updateGaugeCard(CARD_CASE, snap.caseTemp, AMBIENT_YELLOW, AMBIENT_RED);
    updateGaugeCard(CARD_PWR,  snap.pcPower,  POWER_YELLOW,   POWER_RED);
  }
  char buf[32];
  if (isfinite(snap.cpuTemp)) {
    snprintf(buf, sizeof(buf), "CPU: %.1f°C", snap.cpuTemp);
    lv_label_set_text(ui.legCpuLbl, buf);
    lv_chart_set_next_value(ui.chart, ui.serCpu, (lv_coord_t)roundf(snap.cpuTemp));
  }
  if (isfinite(snap.gpuTemp)) {
    snprintf(buf, sizeof(buf), "GPU: %.1f°C", snap.gpuTemp);
    lv_label_set_text(ui.legGpuLbl, buf);
    lv_chart_set_next_value(ui.chart, ui.serGpu, (lv_coord_t)roundf(snap.gpuTemp));
  }
  if (isfinite(snap.caseTemp)) {
    snprintf(buf, sizeof(buf), "Case: %.1f°C", snap.caseTemp);
    lv_label_set_text(ui.legCaseLbl, buf);
    lv_chart_set_next_value(ui.chart, ui.serCase, (lv_coord_t)roundf(snap.caseTemp));
  }
}
// 11. SETUP & LOOP
void setup() {
  Serial.begin(115200);
  prefs.begin("repos_cfg", false);
  currentBrightness = constrain((int)prefs.getUChar("bright", 100), 5, 100);
  currentHue        = prefs.getUShort("hue", 142) % 360;
  currentSat        = prefs.getUChar("sat", 70);
  if (currentSat > 100) currentSat = 70;
#if defined(ESP_ARDUINO_VERSION_MAJOR) && ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcAttach(BACKLIGHT_PIN, 5000, 8);
#else
  ledcSetup(0, 5000, 8);
  ledcAttachPin(BACKLIGHT_PIN, 0);
#endif
  applyBacklightPwm(currentBrightness);
  lcd.init();
  lcd.setRotation(3);
  lv_init();
  size_t bufPixels = 480 * 40;
  buf1 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  buf2 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  if (!buf1 || !buf2) {
    if (buf1) free(buf1);
    if (buf2) free(buf2);
    bufPixels = 480 * 24;
    buf1 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
    buf2 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  }
  if (!buf1 || !buf2) {
    if (buf1) free(buf1);
    if (buf2) free(buf2);
    bufPixels = 480 * 16;
    buf1 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
    buf2 = (lv_color_t*)heap_caps_malloc(bufPixels * sizeof(lv_color_t), MALLOC_CAP_DMA);
  }
  if (!buf1 || !buf2) {
    Serial.println("LVGL DMA buffer alloc failed!");
    while (true) delay(1000);
  }
  Serial.printf("LVGL DMA buffer: %u lines/buf\n", (unsigned)(bufPixels / 480));
  lv_disp_draw_buf_init(&draw_buf, buf1, buf2, bufPixels);
  static lv_disp_drv_t disp_drv;
  lv_disp_drv_init(&disp_drv);
  disp_drv.hor_res  = 480;
  disp_drv.ver_res  = 320;
  disp_drv.flush_cb = my_disp_flush;
  disp_drv.draw_buf = &draw_buf;
  lv_disp_t* disp = lv_disp_drv_register(&disp_drv);
  if (disp && disp->refr_timer) {
    lv_timer_set_period(disp->refr_timer, 20);
  }
  static lv_indev_drv_t indev_drv;
  lv_indev_drv_init(&indev_drv);
  indev_drv.type         = LV_INDEV_TYPE_POINTER;
  indev_drv.read_cb      = my_touchpad_read;
  indev_drv.scroll_limit = 16; // Chống hủy bấm khi tay hơi rung nhẹ
  lv_indev_t* indev = lv_indev_drv_register(&indev_drv);
  if (indev && indev->driver && indev->driver->read_timer) {
    lv_timer_set_period(indev->driver->read_timer, 10);
  }
  buildUI();
  bootStartMs = millis();
  dataMutex = xSemaphoreCreateMutex();
  xTaskCreatePinnedToCore(networkTask, "net_task", 8192, NULL, 1, NULL, 0);
}
void loop() {
#if !LV_TICK_CUSTOM
  static uint32_t lastTick = 0;
  uint32_t now = millis();
  lv_tick_inc(now - lastTick);
  lastTick = now;
#endif
  updateSplash();
  updateClock();
  updateUI();
  lv_timer_handler();
  delay(1);
}
